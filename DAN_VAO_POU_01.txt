(* Class result mailbox: commit, slot, sequence, class, session, generation.
   Late results cannot overwrite another product or another session. *)
IF D110 <> D43 THEN
    IF (D111 >= K0) AND (D111 < K16) AND (D114 = D190) THEN
        Z0 := D111 * K8;
        IF (D1000Z0 = D112) AND (D1007Z0 = D115)
           AND (D1002Z0 = K1) AND (D113 >= K1) AND (D113 <= K3) THEN
            D1001Z0 := D113;
        END_IF;
    END_IF;
    D43 := D110;
END_IF;

(* Completion ACK, only after the PC has durably saved the record. *)
IF D120 <> D44 THEN
    IF (D121 >= K0) AND (D121 < K16) AND (D123 = D190) THEN
        Z0 := D121 * K8;
        IF (D1000Z0 = D122) AND (D1007Z0 = D124)
           AND ((D1002Z0 = K5) OR (D1002Z0 = K6) OR (D1002Z0 = K8)) THEN
            FMOV(TRUE, K0, K8, D1000Z0);
        END_IF;
    END_IF;
    D44 := D120;
END_IF;

(* Remove expired queue HEADS before publishing timeout completion.
   D5 = 500 ms stable clear to rearm sensors; D6 = 30 s BELT MOTION
   allowed to reach the next sensor. Stopped time never ages a queue.
   Stage 8 means sensor passage was missed; it is not a measured exit.
   A slot is removed from its queue BEFORE PC may ACK/reuse it. *)
FOR Z2 := K0 TO K15 BY K1 DO
    Z1 := D30;
    Z0 := D1200Z1 * K8;
    D64 := D41 - D1005Z0;
    MOV(D64 < K0, D64 + K32760, D64);
    M64 := (D32 > K0) AND (D1002Z0 = K1) AND (D64 >= D6);
    M65 := (D32 > K0) AND ((D1002Z0 <> K1) OR M64);
    MOV(M65, (D30 + K1) MOD K16, D30);
    MOV(M65, D32 - K1, D32);
    MOV(M64, D39, D1004Z0);
    MOV(M64, D45, D1005Z0);
    MOV(M64, K4, D1006Z0);
    MOV(M64, K8, D1002Z0);
    Z1 := D33;
    Z0 := D1220Z1 * K8;
    D65 := D41 - D1005Z0;
    MOV(D65 < K0, D65 + K32760, D65);
    M66 := (D35 > K0) AND (D1002Z0 = K2) AND (D65 >= D6);
    M67 := (D35 > K0) AND ((D1002Z0 <> K2) OR M66);
    MOV(M67, (D33 + K1) MOD K16, D33);
    MOV(M67, D35 - K1, D35);
    MOV(M66, D39, D1004Z0);
    MOV(M66, D45, D1005Z0);
    MOV(M66, K4, D1006Z0);
    MOV(M66, K8, D1002Z0);
END_FOR;

(* Timing fix D1 - 2026-10-18: native 100 ms timers T4/T5.
   D1=K10 is 1 second, independent of sampled M8011 rising edges.
   Read TN4/TN5 BEFORE a disabled OUT_T resets them. D60/D61 retain
   completed 100 ms units during STOP / another station's belt pause;
   D62/D63 are the remaining preset. Pause quantization is under 100 ms.
   T4/T5 reset outside phase 1; each NEW station job clears D60/D61.
   Extension/retraction timers reset when STOP is pressed.
   STOP during retraction -> repeat extension on the next START. *)
